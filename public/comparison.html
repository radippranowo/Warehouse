Contoh Real Code

❌ Without API (Old)
// Controller
public function pemasukan()
{
    return Inertia::render('Form', [
        'barangs' => Barang::with([
            'kategori',
            'merk',
            'group'
        ])->get(), // Load 1000+ items!
    ]);
}

// Vue Component
<SearchSelect
  :options="barangs" // 1000+ items di props!
  optionValue="id"
  optionLabel="nama_barang"
/>

// Result:
// - Page load: 2-3 seconds
// - Memory: ~50MB
// - Dropdown lag
✅ With API (New)
// Controller
public function pemasukan()
{
    return Inertia::render('Form', [
        // Tidak load barangs!
    ]);
}

// Vue Component
<SearchSelectApi
  api-endpoint="/api/v1/barang/search"
  placeholder="Ketik untuk cari..."
  @selected="onSelected"
/>

// Result:
// - Page load: 0.3 seconds (10x faster!)
// - Memory: ~5MB (90% less!)
// - Autocomplete instant (cached!)
